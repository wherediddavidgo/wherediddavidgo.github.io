def forward_regression(x, y, alpha):
    remaining_predictors = list(x.columns)
    selected_predictors = []

    while len(remaining_predictors) > 0:
        pvals = pd.Series(index=remaining_predictors, dtype=float)

        for n in range(len(remaining_predictors)):
            candidate = remaining_predictors[n]
            x_model = sm.add_constant(x[selected_predictors + [candidate]])
            y_model = sm.OLS(y, x_model).fit()
            pvals[candidate] = y_model.pvalues[candidate]
        
        best_pval = np.min(pvals)

        if best_pval < alpha:
            best_feature = pvals.idxmin()
            selected_predictors.append(best_feature)
            remaining_predictors.remove(best_feature)
        else:
            break
    
    result = sm.OLS(y, sm.add_constant(x[selected_predictors])).fit()

    return result

def backward_regression(x, y, alpha):
    remaining_predictors = list(x.columns)
    selected_predictors = []

    while len(remaining_predictors) > 0:
        pvals = pd.Series(index=remaining_predictors, dtype=float)

        for n in range(len(remaining_predictors)):
            candidate = remaining_predictors[n]
            x_model = sm.add_constant(x[selected_predictors + [candidate]])
            y_model = sm.OLS(y, x_model).fit()
            pvals[candidate] = y_model.pvalues[candidate]
        
        worst_pval = np.max(pvals)

        if worst_pval > alpha:
            worst_feature = pvals.idxmax()
            remaining_predictors.remove(worst_feature)
        else:
            break
    
    result = sm.OLS(y, sm.add_constant(x[remaining_predictors])).fit()

    return result
